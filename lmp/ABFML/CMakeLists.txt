cmake_minimum_required(VERSION 3.18)
project(abfml LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
set(LAMMPS_DIR "" CACHE PATH "LAMMPS source root")
find_path(LAMMPS_INCLUDE_DIR pair.h PATHS "${LAMMPS_DIR}/src" REQUIRED)
find_library(LAMMPS_LIBRARY NAMES lammps PATHS "${LAMMPS_DIR}/build" "${LAMMPS_DIR}/build/src" REQUIRED)
add_library(abfml MODULE pair_abfml.cpp plugin.cpp)
target_include_directories(abfml PRIVATE "${LAMMPS_INCLUDE_DIR}")
target_link_libraries(abfml PRIVATE "${LAMMPS_LIBRARY}" MPI::MPI_CXX pybind11::embed)
set_target_properties(abfml PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE)
